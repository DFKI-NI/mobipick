cmake_minimum_required(VERSION 3.0.2)
project(ur_polished_kinematics)
find_package(catkin REQUIRED COMPONENTS roscpp geometry_msgs moveit_core moveit_kinematics moveit_ros_planning pluginlib tf_conversions kdl_parser)
find_package(Boost REQUIRED)
catkin_package(INCLUDE_DIRS include LIBRARIES ur5_polished_kinematics CATKIN_DEPENDS roscpp geometry_msgs moveit_core moveit_kinematics moveit_ros_planning pluginlib tf_conversions kdl_parser DEPENDS Boost)
include_directories(include ${catkin_INCLUDE_DIRS} ${Boost_INCLUDE_DIRS})
add_library(ur5_polished_kinematics src/ur_kin.cpp src/ur_moveit_plugin.cpp)
target_compile_definitions(ur5_polished_kinematics PRIVATE UR5_PARAMS)
target_link_libraries(ur5_polished_kinematics ${catkin_LIBRARIES} ${Boost_LIBRARIES})
install(TARGETS ur5_polished_kinematics LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES ur_moveit_plugin.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
